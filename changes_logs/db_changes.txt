  `id` int(11) NOT NULL,
  `email_type` varchar(100) NOT NULL,
  `email_data` mediumtext NOT NULL,
  `status` tinyint(4) NOT NULL DEFAULT '0',
  `updated_by` int(11) NOT NULL,
  `created` datetime NOT NULL,
  `modified` timestamp NOT NULL 
) ENGINE=InnoDB DEFAULT CHARSET=latin1;


DELIMITER $$
CREATE TRIGGER `email_queues_delete_trigger` BEFORE DELETE ON `email_queues` FOR EACH ROW BEGIN
  INSERT INTO email_queues_audit select * from email_queues where id = OLD.id;
  UPDATE email_queues_audit SET deleted=now(), status=1 where id = OLD.id;
END
$$
DELIMITER ;
DELIMITER $$
CREATE TRIGGER `email_queues_update_trigger` BEFORE UPDATE ON `email_queues` FOR EACH ROW BEGIN
  INSERT INTO email_queues_audit select * from email_queues where id = NEW.id;
END
$$
DELIMITER ;
